import '../../scripts/offline_guard.cjs';
import assert from 'node:assert/strict';
import { MarketContextManager } from '../src/marketData/marketContext.js';
import { ChartSession } from '../src/session.js';
import { initializeOrdinaryRuntime } from '../src/marketData/ordinaryRuntime.js';
initializeOrdinaryRuntime({FUTURES_PROVIDER:'none'});
let starts=0,closes=0; const messages:any[]=[];
const factory:any=(symbol:string,_i:any,h:any)=>({provider:'phase06-fixture',feed:{provider:'phase06-fixture',symbol,connect:async()=>{starts++;h.onStatus({state:'CONNECTING',provider:'phase06-fixture',symbol});},disconnect:async()=>{closes++;},isConnected:()=>true}});
const manager=new MarketContextManager(undefined,undefined,undefined,factory);
const session=new ChartSession({readyState:1,bufferedAmount:0,send:(x:string)=>messages.push(JSON.parse(x)),close:()=>{}} as any);
session.setUser({id:'offline',username:'offline',role:'admin',status:'active'});
await manager.subscribe(session,'ES','1m',session.nextGeneration());
await (manager.subscribe as any)(session,'NQ','1m',session.nextGeneration(),'pane2');
const before={primary:session.subscribedSymbol,ES:manager.getContext('ES')!.subscriberCount,NQ:manager.getContext('NQ')!.subscriberCount,subscriptions:[...session.subscriptions.keys()]};
manager.unsubscribe(session);
const after={ES:manager.getContext('ES')!.subscriberCount,NQ:manager.getContext('NQ')!.subscriberCount,subscriptions:[...session.subscriptions.keys()]};
console.log(JSON.stringify({scope:'OFFLINE PROBE — NOT ACCEPTANCE PASS',expected:{primary:'ES',afterClose:{ES:0,NQ:0,subscriptions:[]}},actual:{before,after},starts,closes}));
await manager.disposeAll();
assert.equal(starts,2);process.exit(0);
